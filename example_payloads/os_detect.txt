REM AUTHOR: Korben

REM Generic OS Detection for Windows, Mac, Linux
REM Advanced Detection includes iOS, ChromeOS, Android

REM CONFIGURATION
DEFINE DEBUGGING_OUTPUT FALSE
DEFINE ADVANCED_DETECTION FALSE
DEFINE STARTUP_DELAY 1500
DEFINE RESTART_WAIT 1000
DEFINE CONNECT_WAIT 1000
DEFINE OS_DETECT_MODE HID
DEFINE OS_DETECT_VID VID_05AC
DEFINE OS_DETECT_PID PID_021E
DEFINE WINDOWS_HOST_REQUEST_COUNT 2
DEFINE HOST_RESPONSE_TIMEOUT 1000

REM START OS DETECTION FUNCTION
DELAY STARTUP_DELAY
ATTACKMODE OS_DETECT_MODE OS_DETECT_VID OS_DETECT_PID
SAVE_HOST_KEYBOARD_LOCK_STATE

IF DEBUGGING_OUTPUT THEN
  IF ADVANCED_DETECTION THEN
    STRING "ADVANCED OS DETECT"
  ELSE
    STRING "OS DETECT"
  ENDIF
  ENTER
  STRING "test caps"
ENDIF

IF CAPSLOCK OFF THEN
  LED R
  CAPSLOCK
  DELAY HOST_RESPONSE_TIMEOUT
ENDIF
LED OFF

IF DEBUGGING_OUTPUT THEN
  ENTER
  STRING "test done"
ENDIF

IF RECEIVED_HOST_LOCK_LED_REPLY THEN
  IF DEBUGGING_OUTPUT THEN
    ENTER
    STRING "received led response"
  ENDIF
  LED G
  IF $_HOST_CONFIGURATION_REQUEST_COUNT > WINDOWS_HOST_REQUEST_COUNT THEN
    IF DEBUGGING_OUTPUT THEN
      ENTER
      STRING "prediction: Windows"
    ENDIF
    DEFINE OS WINDOWS
  ELSE
    IF DEBUGGING_OUTPUT THEN
      ENTER
      STRING "prediction: Linux"
    ENDIF
    DEFINE OS LINUX
  ENDIF
ELSE
  IF DEBUGGING_OUTPUT THEN
    ENTER
    STRING "no led response"
    ENTER
    STRING "prediction: MacOS"
  ENDIF
  DEFINE OS MACOS
ENDIF

REM ADVANCED DETECTION LOGIC
IF ADVANCED_DETECTION THEN
  IF OS == LINUX THEN
    IF DEBUGGING_OUTPUT THEN
      ENTER
      STRING "soft reconnect"
    ENDIF
    ATTACKMODE OFF
    DELAY RESTART_WAIT
    ATTACKMODE OS_DETECT_MODE OS_DETECT_VID OS_DETECT_PID
    DELAY CONNECT_WAIT
    IF DEBUGGING_OUTPUT THEN
      ENTER
      STRING "reconnected"
    ENDIF
    IF CAPSLOCK ON THEN
      IF DEBUGGING_OUTPUT THEN
        ENTER
        STRING "caps led on"
        ENTER
        STRING "test numlock"
      ENDIF
      NUMLOCK
      DELAY HOST_RESPONSE_TIMEOUT
      IF DEBUGGING_OUTPUT THEN
        ENTER
        STRING "test done"
      ENDIF
      IF NUMLOCK OFF THEN
        IF DEBUGGING_OUTPUT THEN
          ENTER
          STRING "no numlock led"
          ENTER
          STRING "prediction: ChromeOS"
        ENDIF
        DEFINE OS CHROMEOS
      ELSE
        IF DEBUGGING_OUTPUT THEN
          ENTER
          STRING "numlock led on"
          ENTER
          STRING "testing scrolllock"
        ENDIF
        SCROLLLOCK
        DELAY HOST_RESPONSE_TIMEOUT
        IF DEBUGGING_OUTPUT THEN
          ENTER
          STRING "test done"
        ENDIF
        IF SCROLLLOCK ON THEN
          IF DEBUGGING_OUTPUT THEN
            ENTER
            STRING "scrolllock led on"
            ENTER
            STRING "prediction: Android"
          ENDIF
          DEFINE OS ANDROID
        ELSE
          IF DEBUGGING_OUTPUT THEN
            ENTER
            STRING "no scrolllock reply"
            ENTER
            STRING "prediction: Linux"
          ENDIF
          DEFINE OS LINUX
        ENDIF
      ENDIF
    ENDIF
  ELSEIF OS == MACOS THEN
    IF CAPSLOCK ON THEN
      IF DEBUGGING_OUTPUT THEN
        ENTER
        STRING "caps led on"
        ENTER
        STRING "prediction: iOS"
      ENDIF
      DEFINE OS IOS
    ELSE
      IF DEBUGGING_OUTPUT THEN
        ENTER
        STRING "no caps reply"
        ENTER
        STRING "prediction: MacOS"
      ENDIF
      DEFINE OS MACOS
    ENDIF
  ELSEIF OS == WINDOWS THEN
    IF DEBUGGING_OUTPUT THEN
      ENTER
      STRING "Confident Windows Prediction"
    ENDIF
    DEFINE OS WINDOWS
  ENDIF
ENDIF

RESTORE_HOST_KEYBOARD_LOCK_STATE

IF DEBUGGING_OUTPUT THEN
  ENTER
  STRING "OS_DETECT complete"
  ENTER
ENDIF
REM END OS DETECTION FUNCTION
